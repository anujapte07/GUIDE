    wire [7:0] sum;
    wire       cout, ovf;

    reg  [8:0] exp_full;
    reg        exp_ovf;
    integer    tests, errors;

    adder8 dut (.a(a), .b(b), .cin(cin), .sum(sum), .cout(cout), .ovf(ovf));

    // Apply one vector, wait for the ripple to settle, compare against the reference model.

        input [7:0] ta;
        input [7:0] tb;

            #1;

        end
